# googlesearch response

Layout 구조를 **"왼쪽에는 고정된 달력(Calendar), 오른쪽에는 일기 리스트(List)"**가 나오는 2단 분할 구조(Split View)로 변경하고 싶으시군요!

C++로 비유하자면 UI 창의 레이아웃을 수평 분할(Horizontal Splitter) 구조로 바꾸는 과정입니다. HTML5의 네이티브 달력은 가로로 길게 나오는 특성이 있어, 왼쪽 공간에 정사각형 모양으로 예쁘게 박히도록 CSS 디자인과 HTML 구조를 연동하여 완전히 개조해 드리겠습니다.

**`src/App.tsx`** 파일을 열고 전체 코드를 아래 코드로 **통째로 덮어쓰기** 한 뒤 저장해 보세요!

📝 좌우 2단 분할 구조 적용 `src/App.tsx` 전체 소스 코드

```tsx
import { useState, useEffect } from "react";
import { invoke } from "@tauri-apps/api/core";
import "./App.css";

interface DiaryItem {
  id: number;
  date: string;
  content: string;
}

function App() {
  const [diaries, setDiaries] = useState<DiaryItem[]>([]);
  const [inputText, setInputText] = useState("");
  const [selectedDate, setSelectedDate] = useState("");

  // 앱 구동 시 데이터 로드
  useEffect(() => {
    const fetchDiaries = async () => {
      try {
        const lines: string[] = await invoke("load_diaries_from_file");

        const parsedDiaries: DiaryItem[] = lines.map((line, index) => {
          const closeBracketIndex = line.indexOf("]");
          if (closeBracketIndex !== -1 && line.startsWith("[")) {
            const date = line.substring(1, closeBracketIndex);
            const content = line.substring(closeBracketIndex + 2);
            return { id: Date.now() + index, date, content };
          }
          return { id: Date.now() + index, date: "알 수 없는 날짜", content: line };
        });

        setDiaries(parsedDiaries.reverse());
      } catch (error) {
        console.error("데이터 로드 실패:", error);
      }
    };

    fetchDiaries();
  }, []);

  // 일기 추가 핸들러
  const handleAddDiary = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!inputText.trim()) return;

    const now = new Date();
    const formattedDate = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(now.getDate()).padStart(2, '0')} ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

    try {
      await invoke("save_diary_to_file", {
        date: formattedDate,
        content: inputText
      });

      const newDiary: DiaryItem = {
        id: Date.now(),
        date: formattedDate,
        content: inputText,
      };

      setDiaries([newDiary, ...diaries]);
      setInputText("");
    } catch (error) {
      alert("하드디스크 저장 오류 발생: " + error);
    }
  };

  // 날짜 필터링 로직
  const filteredDiaries = selectedDate
    ? diaries.filter((item) => item.date.startsWith(selectedDate))
    : diaries;

  return (
    <div className="app-window">
      {/* 상단 타이틀 헤더 */}
      <header className="app-header">
        <h1>간단 개발 일기장</h1>
        <p className="subtitle">To-Do 스타일로 가볍게 한 줄씩 기록하세요.</p>
      </header>

      {/* 🚀 [핵심 변경] 좌우 분할 플렉스 레이아웃 컨테이너 */}
      <div className="main-content-layout">

        {/* ⬅️ 왼쪽 섹션: 달력 컨트롤러 고정 배치 */}
        <aside className="left-sidebar">
          <div className="calendar-card">
            <h3>📅 날짜별 조회</h3>
            <input
              type="date"
              value={selectedDate}
              onChange={(e) => setSelectedDate(e.target.value)}
              className="date-picker-sidebar"
            />
            <button
              onClick={() => setSelectedDate("")}
              className="clear-filter-btn"
              disabled={!selectedDate}
            >
              전체 목록 보기
            </button>
          </div>
        </aside>

        {/* ➡️ 오른쪽 섹션: 입력창 및 일기 출력 리스트 */}
        <main className="right-content">
          {/* 입력창 영역 */}
          <form onSubmit={handleAddDiary} className="input-form">
            <input
              type="text"
              value={inputText}
              onChange={(e) => setInputText(e.target.value)}
              placeholder="오늘 어떤 개발을 하셨나요? (엔터를 치면 등록됩니다)"
              className="diary-input"
            />
            <button type="submit" className="submit-btn">등록</button>
          </form>

          {/* 일기 리스트 출력 영역 */}
          <div className="diary-list">
            {filteredDiaries.length === 0 ? (
              <p className="empty-message">해당 조건에 작성된 기록이 없습니다.</p>
            ) : (
              filteredDiaries.map((item) => (
                <div key={item.id} className="diary-card">
                  <div className="diary-date">{item.date}</div>
                  <div className="diary-content">{item.content}</div>
                </div>
              ))
            )}
          </div>
        </main>

      </div>
    </div>
  );
}

export default App;
```
