📝 `src/App.tsx` 전체 소스 코드

```tsx
import { useState } from "react";
import "./App.css";

// 일기 한 건의 데이터 구조 정의 (C++의 struct 역할)
interface DiaryItem {
  id: number;
  date: string;
  content: string;
}

function App() {
  // 상태 관리 (변수 값이 바뀌면 화면을 알아서 다시 그려주는 React의 핵심 기능)
  const [diaries, setDiaries] = useState<DiaryItem[]>([]);
  const [inputText, setInputText] = useState("");

  // 엔터를 치거나 등록 버튼을 눌렀을 때 실행되는 함수
  const handleAddDiary = (e: React.FormEvent) => {
    e.preventDefault(); // 페이지 새로고침 방지

    if (!inputText.trim()) return; // 빈 칸이면 무시

    // 현재 날짜와 시간 구하기 (자동 입력)
    const now = new Date();
    const formattedDate = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}-${String(now.getDate()).padStart(2, '0')} ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

    // 새 일기 객체 생성
    const newDiary: DiaryItem = {
      id: Date.now(), // 겹치지 않는 고유 ID 값
      date: formattedDate,
      content: inputText,
    };

    // 기존 일기 배열 뒤에 새 일기 추가 (Vector에 push_back 하는 느낌)
    setDiaries([newDiary, ...diaries]);
    setInputText(""); // 입력창 비우기
  };

  return (
    <div className="container">
      <h1>간단 개발 일기장</h1>
      <p className="subtitle">To-Do 스타일로 가볍게 한 줄씩 기록하세요.</p>

      {/* 입력창 영역 */}
      <form onSubmit={handleAddDiary} className="input-form">
        <input
          type="text"
          value={inputText}
          onChange={(e) => setInputText(e.target.value)}
          placeholder="오늘 어떤 개발을 하셨나요? (엔터를 치면 등록됩니다)"
          className="diary-input"
        />
        <button type="submit" className="submit-btn">등록</button>
      </form>

      {/* 일기 리스트 출력 영역 (C++의 for문 도는 느낌) */}
      <div className="diary-list">
        {diaries.length === 0 ? (
          <p className="empty-message">아직 작성된 기록이 없습니다.</p>
        ) : (
          diaries.map((item) => (
            <div key={item.id} className="diary-card">
              <div className="diary-date">{item.date}</div>
              <div className="diary-content">{item.content}</div>
            </div>
          ))
        )}
      </div>
    </div>
  );
}

export default App;
```
